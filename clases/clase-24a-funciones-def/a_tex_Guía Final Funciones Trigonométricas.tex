\subsection{1. Medidas angulares}

En esta guía trabajaremos principalmente con \textbf{radianes}. Las equivalencias fundamentales son:
\[
180^\circ=\pi \text{ rad}
\qquad\text{y}\qquad
360^\circ=2\pi \text{ rad}.
\]

\begin{formulabox}[Conversión rápida: regla de tres]
Para transformar grados a radianes:
\[
\frac{\alpha^\circ}{180^\circ}=\frac{x}{\pi}
\qquad\Longrightarrow\qquad
x=\frac{\alpha\pi}{180}.
\]
\end{formulabox}

\begin{ejemplobox}[Ejemplo breve]
Convertir $60^\circ$ a radianes:
\[
x=\frac{60\pi}{180}=\frac{\pi}{3}.
\]
\end{ejemplobox}

\begin{center}
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{c|cccccc}
\toprule
$\alpha$ en grados & $0^\circ$ & $30^\circ$ & $45^\circ$ & $60^\circ$ & $90^\circ$ & $180^\circ$ \\
\midrule
$\alpha$ en radianes & $0$ & $\frac{\pi}{6}$ & $\frac{\pi}{4}$ & $\frac{\pi}{3}$ & $\frac{\pi}{2}$ & $\pi$ \\
$\sen\alpha$ & $0$ & $\frac12$ & $\frac{\sqrt2}{2}$ & $\frac{\sqrt3}{2}$ & $1$ & $0$ \\
$\cos\alpha$ & $1$ & $\frac{\sqrt3}{2}$ & $\frac{\sqrt2}{2}$ & $\frac12$ & $0$ & $-1$ \\
\bottomrule
\end{tabular}
\end{center}

\subsection{2. Circunferencia goniométrica, signos y ángulo de referencia}

En la circunferencia de radio $1$, el punto asociado a un ángulo $\alpha$ tiene coordenadas
\[
\boxed{P=(\cos\alpha,\sen\alpha)}.
\]
Por ello, $\cos\alpha$ se lee en el eje $x$ y $\sen\alpha$ en el eje $y$; ambos toman valores entre $-1$ y $1$.

\begin{center}

    \draw[-{Latex}] (-1.25,0) -- (1.25,0) node[right]{$x$};
    \draw[-{Latex}] (0,-1.25) -- (0,1.25) node[above]{$y$};
    \draw[thick] (O) circle (1);
    \coordinate (P) at (40:1);
    \coordinate (footX) at ({cos(40)},0);
    \coordinate (footY) at (0,{sin(40)});
    \draw[thick,blue] (O) -- (P);
    \draw[dashed,gray] (P) -- (footX);
    \draw[dashed,gray] (P) -- (footY);
    \fill (P) circle (0.7pt);
    \node[above right] at (P) {\footnotesize $P(\cos\alpha,\sen\alpha)$};
    \node[below] at (footX) {\footnotesize $\cos\alpha$};
    \node[left] at (footY) {\footnotesize $\sen\alpha$};
    \draw (0.28,0) arc (0:40:0.28);
    \node at (0.42,0.15) {\footnotesize $\alpha$};
\end{tikzpicture}
\end{center}
\vspace{-0.25cm}

\noindent
\begin{minipage}{\textwidth}
\textbf{Signo y ángulo de referencia.}
Fuera del primer cuadrante: identifica primero el \textbf{signo} y luego el \textbf{ángulo de referencia} $\alpha'$.

\begin{center}
\renewcommand{\arraystretch}{1.05}
\small
\begin{tabular}{ccccc}
\toprule
Cuadrante & Intervalo & $\sen$ & $\cos$ & Ángulo de referencia \\
\midrule
I & $0<\alpha<\frac{\pi}{2}$ & $+$ & $+$ & $\alpha'=\alpha$ \\
II & $\frac{\pi}{2}<\alpha<\pi$ & $+$ & $-$ & $\alpha'=\pi-\alpha$ \\
III & $\pi<\alpha<\frac{3\pi}{2}$ & $-$ & $-$ & $\alpha'=\alpha-\pi$ \\
IV & $\frac{3\pi}{2}<\alpha<2\pi$ & $-$ & $+$ & $\alpha'=2\pi-\alpha$ \\
\bottomrule
\end{tabular}
\normalsize
\end{center}
\vspace{-0.15cm}

\begin{ejemplobox}[Ejemplo breve]
\small
\[
\frac{5\pi}{4}\in\text{III}
\quad\Longrightarrow\quad
\sen\left(\frac{5\pi}{4}\right)\text{ es negativo},
\qquad
\alpha'=\frac{5\pi}{4}-\pi=\frac{\pi}{4}.
\]
\vspace{-0.2cm}
\[
\therefore\quad
\sen\left(\frac{5\pi}{4}\right)
=-\sen\left(\frac{\pi}{4}\right)
=-\frac{\sqrt2}{2}.
\]
\normalsize
\end{ejemplobox}
\end{minipage}

\subsection{4. Periodicidad: períodos completos + resto}

Seno y coseno son \textbf{funciones periódicas}: repiten sus valores cada $2\pi$. Por eso, podemos quitar cualquier cantidad de períodos completos de $2\pi$ y trabajar solo con el ángulo que sobra.

\[
\sen(\theta+2\pi)=\sen\theta,
\qquad
\cos(\theta+2\pi)=\cos\theta.
\]

\begin{estrategiabox}[Idea clave]
Escribimos el ángulo como
\[
\boxed{
\theta=
\underbrace{2\pi q}_{\text{períodos completos}}
+
\underbrace{r}_{\text{resto que importa}}
}
\]
Entonces, los períodos completos no cambian el valor de la función:
\[
\boxed{\sen\theta=\sen r}
\qquad\text{y}\qquad
\boxed{\cos\theta=\cos r}.
\]
\end{estrategiabox}

\begin{estrategiabox}[Torpedo mental]
\begin{center}
\textbf{Quitar períodos}
$\longrightarrow$
\textbf{ubicar el resto}
$\longrightarrow$
\textbf{signo}
$\longrightarrow$
\textbf{ángulo de referencia}
$\longrightarrow$
\textbf{valor}
\end{center}
\end{estrategiabox}

\newpage
\begin{ejemplobox}[Ejemplo breve]
Evaluar $\sen\left(\frac{21\pi}{4}\right)$.

\textbf{1. Quitamos períodos completos:}
\[
\frac{21\pi}{4}=4\pi+\frac{5\pi}{4}=2(2\pi)+\frac{5\pi}{4}.
\]
Por lo tanto,
\[
\sen\left(\frac{21\pi}{4}\right)=\sen\left(\frac{5\pi}{4}\right).
\]

\textbf{2. Ubicamos el resto:} $\frac{5\pi}{4}$ está en el \textbf{III cuadrante}, por lo tanto el seno es \textbf{negativo}.

\textbf{3. Buscamos el ángulo de referencia:}
\[
\alpha'=\frac{5\pi}{4}-\pi=\frac{\pi}{4}.
\]

\textbf{4. Evaluamos:}
\[
\sen\left(\frac{21\pi}{4}\right)
=-\sen\left(\frac{\pi}{4}\right)
=-\frac{\sqrt2}{2}.
\]
\end{ejemplobox}

\begin{alertabox}[Error frecuente: quitar $\pi$ en vez de $2\pi$]
Las vueltas completas miden $2\pi$. Por ejemplo:
\[
\sen\left(\frac{5\pi}{2}\right)
=\sen\left(\frac{5\pi}{2}-2\pi\right)
=\sen\left(\frac{\pi}{2}\right)=1.
\]
Si restáramos solo $\pi$, obtendríamos
\[
\sen\left(\frac{5\pi}{2}-\pi\right)
=\sen\left(\frac{3\pi}{2}\right)=-1,
\]
que es otro valor.

\medskip
En una función transformada, primero se calcula el \textbf{argumento completo}. Por ejemplo, si $f(x)=\sen(2x)$ y $x=\frac{5\pi}{4}$:
\[
2x=\frac{5\pi}{2},
\]
y recién ese ángulo se reduce por periodicidad.
\end{alertabox}

\subsection{5. Evaluar una función trigonométrica}

Para una función como
\[
f(x)=A\,\sen(Bx-C)+D
\qquad\text{o}\qquad
f(x)=A\cos(Bx-C)+D,
\]
conviene resolver siempre en el mismo orden.

\begin{estrategiabox}[Procedimiento]
\begin{enumerate}[leftmargin=*,itemsep=0.2em]
    \item \textbf{Sustituir} el valor de $x$.
    \item \textbf{Calcular el argumento} completo $\theta=Bx-C$.
    \item \textbf{Quitar períodos completos} de $2\pi$, si es necesario.
    \item \textbf{Evaluar} el seno o coseno usando cuadrante, signo y ángulo de referencia.
    \item \textbf{Aplicar} los valores externos: $A(\text{valor trigonométrico})+D$.
\end{enumerate}
\end{estrategiabox}

\begin{ejemplobox}[Ejemplo breve]
Sea
\[
g(x)=3\sen(x+\pi)+2.
\]
Calcular $g\left(\frac{13\pi}{6}\right)$.

\textbf{1. Sustituimos $x$:}
\[
g\left(\frac{13\pi}{6}\right)
=3\sen\left(\frac{13\pi}{6}+\pi\right)+2.
\]

\textbf{2. Calculamos el argumento:}
\[
\frac{13\pi}{6}+\pi=\frac{19\pi}{6}.
\]

\textbf{3. Quitamos períodos completos:}
\[
\frac{19\pi}{6}=2\pi+\frac{7\pi}{6}
\quad\Longrightarrow\quad
\sen\left(\frac{19\pi}{6}\right)=\sen\left(\frac{7\pi}{6}\right).
\]

\textbf{4. Evaluamos el seno:} $\frac{7\pi}{6}$ está en III, por lo tanto el seno es negativo. Además,
\[
\alpha'=\frac{7\pi}{6}-\pi=\frac{\pi}{6}
\quad\Longrightarrow\quad
\sen\left(\frac{7\pi}{6}\right)=-\frac12.
\]

\textbf{5. Terminamos la función:}
\[
g\left(\frac{13\pi}{6}\right)
=3\left(-\frac12\right)+2
=\frac12.
\]
\end{ejemplobox}

\subsection{6. Transformaciones de seno y coseno}

Usaremos la misma forma general en toda la guía:
\[
\boxed{f(x)=A\,\sen(Bx-C)+D}
\qquad\text{o}\qquad
\boxed{f(x)=A\cos(Bx-C)+D}.
\]

\begin{center}
\renewcommand{\arraystretch}{1.55}
\begin{tabular}{p{0.25\textwidth}p{0.30\textwidth}p{0.34\textwidth}}
\toprule
Qué observar & Qué determina & Regla \\
\midrule
$A$ & Amplitud & $\text{amplitud}=|A|$ \\
$B$ & Período & $P=\dfrac{2\pi}{|B|}$ \\
$\dfrac{C}{B}$ & Desplazamiento horizontal & $>0$: derecha; $<0$: izquierda \\
$D$ & Línea media & $y=D$ \\
\bottomrule
\end{tabular}
\end{center}

Además,
\[
\boxed{\text{máximo}=D+|A|}
\qquad
\boxed{\text{mínimo}=D-|A|}.
\]

Para leer el desplazamiento horizontal, factorizamos $B$:
\[
Bx-C=B\left(x-\frac{C}{B}\right).
\]
Por eso,
\[
\boxed{\text{desplazamiento horizontal}=\frac{C}{B}}.
\]

\begin{ejemplobox}[Ejemplo breve]
Para
\[
f(x)=3\sen\left(2x-\frac{\pi}{2}\right)+1,
\]
tenemos:
\[
\text{amplitud}=3,
\qquad
P=\frac{2\pi}{2}=\pi,
\qquad
\frac{C}{B}=\frac{\pi/2}{2}=\frac{\pi}{4}\text{ a la derecha},
\qquad
D=1.
\]
Por tanto, su máximo es $4$ y su mínimo es $-2$.
\end{ejemplobox}

\begin{alertabox}[Error frecuente]
El signo del desplazamiento horizontal se interpreta al revés del que vemos junto a $x$. Por ejemplo:
\[
x+\pi=x-(-\pi),
\]
por lo tanto el desplazamiento es $-\pi$: \textbf{$\pi$ hacia la izquierda}.
\end{alertabox}

\subsection{7. Representar una gráfica con cinco puntos}

No es necesario memorizar cinco fórmulas. Basta con encontrar el período y recordar el \textbf{patrón} de cada función.

\begin{estrategiabox}[Procedimiento]
\textbf{1.} Calcula el período:
\[
P=\frac{2\pi}{|B|}.
\]

\textbf{2.} Divide un período en cuatro partes iguales:
\[
\boxed{\frac{P}{4}}.
\]
Marca cinco posiciones separadas por $\frac{P}{4}$. Si hay desplazamiento horizontal, comienza desde ese punto.

\textbf{3.} Sigue el patrón:

\begin{center}
\textbf{Seno:}
\quad
media $\rightarrow$ máximo $\rightarrow$ media $\rightarrow$ mínimo $\rightarrow$ media

\medskip
\textbf{Coseno:}
\quad
máximo $\rightarrow$ media $\rightarrow$ mínimo $\rightarrow$ media $\rightarrow$ máximo
\end{center}
\end{estrategiabox}

\begin{ejemplobox}[Ejemplo breve]
Para
\[
f(x)=2\sen x+1,
\]
identificamos $A=2$, $B=1$ y $D=1$. Por lo tanto,
\[
\text{línea media}=D=1,
\]
\[
\text{máximo}=D+|A|=1+2=3,
\qquad
\text{mínimo}=D-|A|=1-2=-1.
\]

Como $B=1$, el período es
\[
P=\frac{2\pi}{|B|}=\frac{2\pi}{1}=2\pi.
\]
Dividimos el período en cuatro partes:
\[
\frac{P}{4}=\frac{2\pi}{4}=\frac{\pi}{2}.
\]
Por tanto, avanzamos en el eje $x$ de $\frac{\pi}{2}$ en $\frac{\pi}{2}$:
\[
0,\quad \frac{\pi}{2},\quad \pi,\quad \frac{3\pi}{2},\quad 2\pi.
\]
Como es seno, seguimos el patrón
\[
\boxed{\text{media}\rightarrow\text{máx.}\rightarrow\text{media}\rightarrow\text{mín.}\rightarrow\text{media}}
\]
y obtenemos los valores
\[
\boxed{1\rightarrow3\rightarrow1\rightarrow-1\rightarrow1}.
\]
\end{ejemplobox}

\subsection{8. Reconocer una función a partir de su gráfica}

Cuando se entrega una gráfica, busca solo la información necesaria para distinguir las alternativas.

\begin{estrategiabox}[Procedimiento]
\begin{enumerate}[leftmargin=*,itemsep=0.15em]
    \item \textbf{Línea media:}
    \[
    D=\frac{\text{máximo}+\text{mínimo}}{2}.
    \]
    \item \textbf{Amplitud:}
    \[
    |A|=\frac{\text{máximo}-\text{mínimo}}{2}.
    \]
    \item \textbf{Período $P$:} distancia horizontal entre dos puntos equivalentes consecutivos.
    \item \textbf{Si es necesario:}
    \[
    |B|=\frac{2\pi}{P}.
    \]
    \item Si después de estos pasos \textbf{más de una alternativa coincide}, revisa el desplazamiento horizontal o el sentido de la curva.
\end{enumerate}
\end{estrategiabox}

\begin{ejemplobox}[Ejemplo breve]
Una gráfica tiene máximo $7$, mínimo $-5$ y período $4\pi$.
Entonces,
\[
D=\frac{7+(-5)}{2}=1,
\qquad
|A|=\frac{7-(-5)}{2}=6,
\]
\[
|B|=\frac{2\pi}{4\pi}=\frac12.
\]
Por lo tanto, buscamos una alternativa con
\[
D=1,
\qquad
|A|=6,
\qquad
|B|=\frac12.
\]
Esto identifica
\[
\boxed{f(x)=6\sen\left(\frac{x}{2}\right)+1}.
\]
\textbf{No es necesario analizar el desplazamiento:} con línea media, amplitud y período ya quedó una sola alternativa posible.
\end{ejemplobox}

\begin{formulabox}[Estrategia PAES: si hay alternativas]
Compara primero \textbf{línea media $D$}, \textbf{amplitud $|A|$} y \textbf{período $P$}. Muchas alternativas se descartan sin reconstruir toda la función.
\end{formulabox}
